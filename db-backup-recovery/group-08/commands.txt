# Before a restore on a new server, recreate the roles first (Part 1 of access-control.sql), because roles are not in a single-database dump.
pg_dump -U grp08_backup -h localhost -p 5432 -Fc -d catering_management_db -f catering_management_db_activity6.dump
sha256sum catering_management_db_activity6.dump
pg_restore --list catering_management_db_activity6.dump
createdb -U postgres -h localhost -p 5432 grp08_restore_test
pg_restore -U postgres -h localhost -p 5432 -d grp08_restore_test --exit-on-error catering_management_db_activity6.dump
psql -U postgres -d grp08_restore_test -f ../../db-access-control/group-08/access-control.sql

# Recovery simulation (only in grp08_restore_test)
