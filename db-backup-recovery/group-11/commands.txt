# Before a restore on a new server, recreate the roles first (Part 1 of access-control.sql), because roles are not in a single-database dump.
pg_dump -U grp11_backup -h localhost -p 5432 -Fc -d ReserveIT -f reserveit_activity6.dump
sha256sum reserveit_activity6.dump
pg_restore --list reserveit_activity6.dump
createdb -U postgres -h localhost -p 5432 grp11_restore_test
pg_restore -U postgres -h localhost -p 5432 -d grp11_restore_test --exit-on-error reserveit_activity6.dump
psql -U postgres -d grp11_restore_test -f ../../db-access-control/group-11/access-control.sql

# Recovery simulation (only in grp11_restore_test)
psql -U postgres -d grp11_restore_test -c "INSERT INTO "department" ("department_name") VALUES ('ACT6-DISPOSABLE')"
pg_dump -U postgres -h localhost -Fc -d grp11_restore_test -f grp11_before_incident.dump

pg_restore -U postgres -h localhost -d grp11_recovery_test --exit-on-error grp11_before_incident.dump
