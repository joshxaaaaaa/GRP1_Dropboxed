# Before a restore on a new server, recreate the roles first (Part 1 of access-control.sql), because roles are not in a single-database dump.
pg_dump -U grp04_backup -h localhost -p 5432 -Fc -d grp03_projectdb -f grp03_projectdb_activity6.dump
sha256sum grp03_projectdb_activity6.dump
pg_restore --list grp03_projectdb_activity6.dump
createdb -U postgres -h localhost -p 5432 grp04_restore_test
pg_restore -U postgres -h localhost -p 5432 -d grp04_restore_test --exit-on-error grp03_projectdb_activity6.dump
psql -U postgres -d grp04_restore_test -f ../../db-access-control/group-04/access-control.sql

# Recovery simulation (only in grp04_restore_test)
psql -U postgres -d grp04_restore_test -c "INSERT INTO "specialties" ("specialty_name") VALUES ('ACT6-DISPOSABLE')"
pg_dump -U postgres -h localhost -Fc -d grp04_restore_test -f grp04_before_incident.dump
psql -U postgres -d grp04_restore_test -c "DELETE FROM "public"."specialties" WHERE "specialty_name" = 'ACT6-DISPOSABLE'"
createdb -U postgres -h localhost grp04_recovery_test
pg_restore -U postgres -h localhost -d grp04_recovery_test --exit-on-error grp04_before_incident.dump
