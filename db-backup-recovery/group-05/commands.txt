# Before a restore on a new server, recreate the roles first (Part 1 of access-control.sql), because roles are not in a single-database dump.
pg_dump -U grp05_backup -h localhost -p 5432 -Fc -d grp10_shioricafedb -f grp10_shioricafedb_activity6.dump
sha256sum grp10_shioricafedb_activity6.dump
pg_restore --list grp10_shioricafedb_activity6.dump
createdb -U postgres -h localhost -p 5432 grp05_restore_test
pg_restore -U postgres -h localhost -p 5432 -d grp05_restore_test --exit-on-error grp10_shioricafedb_activity6.dump
psql -U postgres -d grp05_restore_test -f ../../db-access-control/group-05/access-control.sql

# Recovery simulation (only in grp05_restore_test)
